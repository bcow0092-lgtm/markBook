# M3 EPUB 阅读 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 点开书架上的书能分页阅读——翻页、跳目录、看进度、关掉再打开回到原处。

**Architecture:** 阅读引擎按 §5.7 抽成 `ReaderEngine` 接口，M3 只实现 `EpubEngine`（TXT 侧在 M4）。引擎内部的判断逻辑（百分比估算、拖拽跳转的目标决策、当前章节定位）全部抽成**纯函数**放在 `lib/readerProgress.ts`——epub.js 本身难以单测，把决策从它里面剥出来是让这部分有测试覆盖的唯一现实办法。位置索引（`locations`）走新加的 IPC 通道持久化到 `locations/{id}.json`，首次打开后台生成、未就绪时进度退化为估算值。

**Tech Stack:** Electron 44 + React 19 + Zustand 5 + Tailwind CSS 4 + epub.js 0.3.93。

**Spec:** [技术方案文档.md](../../../技术方案文档.md) —— 本计划实现 §5.2（分页渲染与三个坑）、§5.5（进度持久化时机）、§5.7（阅读引擎接口）、§11.2（阅读页视觉）、§11.3（目录面板）。前序里程碑见 [M1](./2026-09-19-m1-骨架与风险验证.md)、[M2](./2026-09-19-m2-书架.md)。

`src/main/services/locations.ts`：

```ts
import { readFile, writeFile, mkdir } from 'node:fs/promises'
import { join } from 'node:path'
import type { AppContext } from './context'

/**
 * id 是主进程自己生成的 uuid，正常不会带路径分隔符。但仍然校验 —— 这个
 * 函数会拿它拼路径，一旦上游哪天改成用书名之类的东西当 id，拼出来就会
 * 穿越到 locations/ 之外。
 */
function assertSafeId(id: string): void {
  if (id === '' || id.includes('/') || id.includes('\\') || id.includes('..') || id.includes('\0')) {
    throw new Error(`非法的书籍 id：${JSON.stringify(id)}`)
  }
}

const fileOf = (ctx: AppContext, id: string): string => {
  assertSafeId(id)
  return join(ctx.paths.locations, `${id}.json`)
}

/** 读位置索引。尚未生成时返回 null —— 这是正常状态，不是错误。 */
export async function readLocations(ctx: AppContext, id: string): Promise<string | null> {
  // fileOf 放在 try 外面：非法 id 应当抛错，不能被下面的 catch 吞成
  // 「这本书没有索引」
  const file = fileOf(ctx, id)
  try {
    return await readFile(file, 'utf8')
  } catch {
    return null
  }
}

export async function writeLocations(ctx: AppContext, id: string, json: string): Promise<void> {
  const file = fileOf(ctx, id)
  await mkdir(ctx.paths.locations, { recursive: true })
  await writeFile(file, json, 'utf8')
}
```

```ts
import type { ReadingProgress } from '@shared/types'
import type { AppContext } from '../services/context'
import { readLocations, writeLocations } from '../services/locations'

export function registerReaderIpc(ctx: AppContext): void {
  ipcMain.handle(IPC.locationsGet, (_e, id: string) => readLocations(ctx, id))
  ipcMain.handle(IPC.locationsSave, (_e, id: string, json: string) => writeLocations(ctx, id, json))

    // lastReadAt 是书架排序的依据（§3.2），每次落进度都要带上
    await ctx.repo.upsertBook({ ...book, progress, lastReadAt: Date.now() })
```

```ts
/**
 * 阅读进度的判断逻辑。
 *
 * 全部是纯函数、刻意不碰 epub.js —— 引擎本身难以单测（把它拉进 jsdom
 * 一个文件要多花 40 秒），把这些决策剥出来是让这部分有测试覆盖的唯一
 * 现实办法。
 */

/** 位置索引未就绪时，按 spine 位置估算全书百分比 */
export function estimatePercentage(spineIndex: number, spineLength: number): number {
  if (spineLength <= 0) return 0
  return clamp01((spineIndex + 1) / spineLength)
}

export type SeekTarget =
  | { kind: 'percentage'; percentage: number }
  | { kind: 'spine'; index: number }

/**
 * 拖拽跳转该往哪跳。
 *
 * 索引就绪时用百分比精确定位；未就绪时只能按章节比例粗跳 —— 这是可接受
 * 的降级，用户至少能跳到大致位置，而不是拖了没反应。
 */
export function seekTarget(
  percentage: number,
  opts: { locationsReady: boolean; spineLength: number },
): SeekTarget {
  const pct = clamp01(percentage)
  if (opts.spineLength <= 0) return { kind: 'spine', index: 0 }
  if (opts.locationsReady) return { kind: 'percentage', percentage: pct }
  return {
    kind: 'spine',
    index: Math.min(opts.spineLength - 1, Math.floor(pct * opts.spineLength)),
  }
}

/** 目录项 + 它在 spine 里的位置。spine 序号取不到时记 -1。 */
export interface IndexedTocEntry {
  title: string
  spineIndex: number
}

export interface ChapterRef {
  /** 在传入的 toc 数组里的下标；没找到时为 -1 */
  index: number
  title: string
}

/**
 * 当前读到哪一章。
 *
 * 判据是 spine 序号：取所有「起点不晚于当前位置」的目录项里的最后一个。
 * 不用 href 字符串比较 —— 目录里的 href 常带锚点、且字符串序不等于阅读序。
 */
export function findCurrentChapter(toc: IndexedTocEntry[], spineIndex: number): ChapterRef {
  let found: ChapterRef = { index: -1, title: '' }
  for (let i = 0; i < toc.length; i++) {
    const entry = toc[i]
    if (entry.spineIndex < 0) continue
    if (entry.spineIndex > spineIndex) break
    found = { index: i, title: entry.title }
  }
  return found
}

function clamp01(n: number): number {
  if (Number.isNaN(n)) return 0
  return Math.min(1, Math.max(0, n))
}
```

```ts
import type { Book } from '@shared/types'

/**
 * 书架默认排序：读过的按最后阅读时间降序在前，没读过的按加入时间降序在后。
 *
 * 这是「没读过」的定义边界：lastReadAt 为 null 就是没读过，它无论 addedAt
 * 多新都要排到读过的后面 —— 否则刚导入一本书就会把正在读的挤下去。
 */
export function sortByRecentRead(books: Book[]): Book[] {
  return [...books].sort((a, b) => {
    const aRead = a.lastReadAt ?? null
    const bRead = b.lastReadAt ?? null
    if (aRead !== null && bRead !== null) return bRead - aRead
    if (aRead !== null) return -1
    if (bRead !== null) return 1
    return b.addedAt - a.addedAt
  })
}
```

```ts
import ePub from 'epubjs'
import type { Book, ReadingProgress } from '@shared/types'
import {
  estimatePercentage,
  findCurrentChapter,
  seekTarget,
  type IndexedTocEntry,
} from '../../lib/readerProgress'
import type { EngineOptions, ReaderEngine } from './engine'

/** 表格兜底：给每张表包一层可横向滚动的容器（技术方案 §8） */
const TABLE_FALLBACK_CSS = `
.epub-table-scroll { overflow-x: auto; max-width: 100%; }
.epub-table-scroll > table { max-width: none; }
`

export function createEpubEngine(opts: EngineOptions): ReaderEngine {
  let book: ReturnType<typeof ePub> | null = null
  let rendition: ReturnType<ReturnType<typeof ePub>['renderTo']> | null = null
  let toc: IndexedTocEntry[] = []
  let spineLength = 0
  let indexReady = false
  let destroyed = false

  function currentProgress(spineIndex: number, cfi: string): ReadingProgress {
    const chapter = findCurrentChapter(toc, spineIndex)
    const percentage =
      indexReady && book
        ? book.locations.percentageFromCfi(cfi)
        : estimatePercentage(spineIndex, spineLength)
    return {
      location: cfi,
      percentage: Number.isFinite(percentage) ? percentage : 0,
      percentageExact: indexReady,
      chapterIndex: chapter.index,
      chapterTitle: chapter.title,
    }
  }

  /** 位置索引生成完成后，用新拿到的精度重发一次进度 */
  function reemitWithExactPercentage(): void {
    if (!rendition || destroyed) return
    const loc = rendition.currentLocation()
    const cfi = (loc as { start?: { cfi?: string } } | undefined)?.start?.cfi
    const index = (loc as { start?: { index?: number } } | undefined)?.start?.index ?? 0
    if (cfi) opts.callbacks.onRelocated(currentProgress(index, cfi))
  }

  async function setupLocations(): Promise<void> {
    if (!book) return
    const saved = await opts.loadLocations().catch(() => null)
    if (destroyed) return

    if (saved) {
      book.locations.load(saved)
      indexReady = true
      reemitWithExactPercentage()
      return
    }

    // 后台生成，首屏不等它（§5.2）。生成期间进度走 estimatePercentage。
    void book.locations
      .generate(1024)
      .then(async () => {
        if (destroyed || !book) return
        indexReady = true
        reemitWithExactPercentage()
        await opts.saveLocations(book.locations.save()).catch(() => {})
      })
      .catch(() => {
        // 生成失败就一直是估算模式，不该让阅读本身崩掉
      })
  }

  function handleContent(contents: { document: Document }): void {
    const doc = contents.document
    const style = doc.createElement('style')
    style.textContent = TABLE_FALLBACK_CSS
    doc.head?.appendChild(style)

    for (const table of Array.from(doc.querySelectorAll('table'))) {
      if (table.parentElement?.classList.contains('epub-table-scroll')) continue
      const wrap = doc.createElement('div')
      wrap.className = 'epub-table-scroll'
      table.parentNode?.insertBefore(wrap, table)
      wrap.appendChild(table)
    }
  }

  return {
    async load(bookRecord: Book, location) {
      book = ePub(opts.bookUrl)
      rendition = book.renderTo(opts.container, {
        flow: 'paginated',
        spread: 'none',
        width: '100%',
        height: '100%',
      })

      rendition.hooks.content.register(handleContent)

      spineLength = (book.spine as unknown as { length: number }).length ?? 0

      rendition.on('relocated', (loc: unknown) => {
        const start = (loc as { start?: { cfi?: string; index?: number } }).start
        if (!start?.cfi) return
        opts.callbacks.onRelocated(currentProgress(start.index ?? 0, start.cfi))
      })

      await rendition.display(location === null ? undefined : String(location))

      const nav = await book.loaded.navigation
      const flat: Array<{ title: string; href: string }> = []
      const walk = (nodes: Array<{ label?: string; href?: string; subitems?: unknown[] }>): void => {
        for (const n of nodes) {
          if (n.href) flat.push({ title: (n.label ?? '').trim(), href: n.href })
          if (n.subitems) walk(n.subitems as typeof nodes)
        }
      }
      walk(nav.toc as never)

      toc = flat.map((e) => ({
        title: e.title,
        spineIndex: (book!.spine.get(e.href) as unknown as { index?: number } | null)?.index ?? -1,
      }))
      opts.callbacks.onToc(toc)

      void setupLocations()
    },

    prev: () => void rendition?.prev(),
    next: () => void rendition?.next(),

    goToChapter(index) {
      const entry = toc[index]
      if (!entry || entry.spineIndex < 0 || !rendition) return
      // 用 spine 里的 href 而不是目录项原始的 href —— 后者常带锚点
      const section = book?.spine.get(entry.spineIndex) as { href?: string } | undefined
      if (section?.href) void rendition.display(section.href)
    },

    goToLocation(location) {
      void rendition?.display(String(location))
    },

    goToPercentage(percentage) {
      if (!rendition || !book) return
      const target = seekTarget(percentage, { locationsReady: indexReady, spineLength })
      if (target.kind === 'percentage') {
        const cfi = book.locations.cfiFromPercentage(target.percentage)
        if (cfi) void rendition.display(cfi)
        return
      }
      const section = book.spine.get(target.index) as { href?: string } | undefined
      if (section?.href) void rendition.display(section.href)
    },

    resize: () => void rendition?.resize(),
    locationsReady: () => indexReady,

    destroy() {
      destroyed = true
      try {
        book?.destroy()
      } catch {
        /* 从未成功打开 */
      }
      book = null
      rendition = null
    },
  }
}
```

```tsx
import { useRef, useState } from 'react'

interface Props {
  /** 0 ~ 1 */
  value: number
  /** 百分比是否为精确值；false 时显示 ~ 前缀 */
  exact?: boolean
  onPreview(percentage: number): void
  onCommit(percentage: number): void
}

function pctLabel(value: number, exact: boolean): string {
  return `${exact ? '' : '~'}${Math.round(value * 100)}%`
}

/**
 * 可拖拽的底部进度条（§11.2）。
 *
 * 拖动过程只回调 onPreview、松手才 onCommit —— 连续 display() 会把渲染
 * 打爆，而且索引未就绪时每次跳转都要重新分页。
 */
export function ProgressBar({ value, exact = true, onPreview, onCommit }: Props) {
  const ref = useRef<HTMLDivElement>(null)
  const [dragging, setDragging] = useState(false)
  const [preview, setPreview] = useState<number | null>(null)

  const ratioAt = (clientX: number): number => {
    const rect = ref.current?.getBoundingClientRect()
    if (!rect || rect.width === 0) return 0
    return Math.min(1, Math.max(0, (clientX - rect.left) / rect.width))
  }

  /** jsdom 没有 pointer capture，真实环境里它保证拖出元素后仍能收到 move */
  const capture = (el: HTMLElement, pointerId: number | undefined): void => {
    if (pointerId === undefined) return
    try {
      el.setPointerCapture(pointerId)
    } catch {
      /* 环境不支持时忽略：没有 capture 只是拖出边界后丢事件，不影响主流程 */
    }
  }

  const shown = preview ?? value

  return (
    <div className="flex items-center gap-3 px-4 py-2">
      <div
        ref={ref}
        data-testid="progress-bar"
        role="slider"
        aria-label="阅读进度"
        aria-valuemin={0}
        aria-valuemax={100}
        aria-valuenow={Math.round(shown * 100)}
        className="relative h-2 flex-1 cursor-pointer rounded bg-neutral-300"
        onPointerDown={(e) => {
          capture(e.currentTarget, e.pointerId)
          const r = ratioAt(e.clientX)
          setDragging(true)
          setPreview(r)
          onPreview(r)
        }}
        onPointerMove={(e) => {
          if (!dragging) return
          const r = ratioAt(e.clientX)
          setPreview(r)
          onPreview(r)
        }}
        onPointerUp={(e) => {
          if (!dragging) return
          const r = ratioAt(e.clientX)
          setDragging(false)
          setPreview(null)
          onCommit(r)
        }}
      >
        <div
          className="absolute inset-y-0 left-0 rounded bg-[var(--color-reader-accent)]"
          style={{ width: `${shown * 100}%` }}
        />
      </div>
      <span
        data-testid="progress-label"
        className="w-12 text-right text-xs text-[var(--color-reader-muted)]"
      >
        {pctLabel(shown, exact)}
      </span>
    </div>
  )
}
```

```ts
import type { MarkReaderApi, ReadingProgress } from '@shared/types'

export interface ProgressSaver {
  /** 记录最新进度并安排落盘。刻意不返回 promise —— relocated 是高频事件 */
  push(progress: ReadingProgress): void
  /** 立即落盘。退出阅读页、窗口关闭时调 */
  flush(): Promise<void>
}

const DEFAULT_DEBOUNCE_MS = 1000

/**
 * 进度的防抖落盘（技术方案 §5.5）。
 *
 * 1 秒防抖；退出阅读页 / 窗口关闭时由调用方调 flush 立即落盘 —— 只靠防抖
 * 的话，用户翻完最后一页立刻关窗，这次进度就丢了。
 */
export function createProgressSaver(
  api: MarkReaderApi,
  bookId: string,
  opts: { debounceMs?: number } = {},
): ProgressSaver {
  const debounceMs = opts.debounceMs ?? DEFAULT_DEBOUNCE_MS
  let pending: ReadingProgress | null = null
  let timer: ReturnType<typeof setTimeout> | null = null

  async function write(): Promise<void> {
    const toWrite = pending
    pending = null
    if (!toWrite) return
    await api.saveProgress(bookId, toWrite).catch(() => {
      // 落盘失败不该打断阅读。下一次 relocated 会再试
    })
  }

  return {
    push(progress) {
      pending = progress
      if (timer) clearTimeout(timer)
      timer = setTimeout(() => {
        timer = null
        void write()
      }, debounceMs)
    },

    async flush() {
      if (timer) {
        clearTimeout(timer)
        timer = null
      }
      await write()
    },
  }
}
```
